import sys pygame.init() # Ekran boyutu WIDTH, HEIGHT = 640, 480 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Labirent Oyunu") # Renkler NIGHT_BLUE = (25, 25, 112) PINK = (255, 105, 180) METAL = (192, 192, 192) GOLD = (255, 215, 0) WHITE = (255, 255, 255) BLACK = (0, 0, 0) # Oyuncu (metal top) player_size = 20 player = pygame.Rect(50, 50, player_size, player_size) # Labirent duvarları (pembe) walls = [ pygame.Rect(100, 100, 400, 20), pygame.Rect(100, 200, 20, 200), pygame.Rect(200, 300, 300, 20) ] # Yıldızlar stars = [ pygame.Rect(150, 150, 15, 15), pygame.Rect(250, 250, 15, 15), pygame.Rect(350, 150, 15, 15) ] # Kara delik (tuzak) blackhole = pygame.Rect(300, 350, 30, 30) # Kupa trophy = pygame.Rect(580, 400, 30, 30) clock = pygame.time.Clock() running = True score = 0 while running: screen.fill(NIGHT_BLUE) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False keys = pygame.key.get_pressed() if keys[pygame.K_LEFT]: player.x -= 5 if keys[pygame.K_RIGHT]: player.x += 5 if keys[pygame.K_UP]: player.y -= 5 if keys[pygame.K_DOWN]: player.y += 5 # Çizimler pygame.draw.rect(screen, METAL, player) for wall in walls: pygame.draw.rect(screen, PINK, wall) for star in stars: pygame.draw.rect(screen, WHITE, star) pygame.draw.rect(screen, BLACK, blackhole) pygame.draw.rect(screen, GOLD, trophy) # Çarpışmalar for wall in walls: if player.colliderect(wall): player.x, player.y = 50, 50 for star in stars[:]: if player.colliderect(star): stars.remove(star) score += 1 if player.colliderect(blackhole): player.x, player.y = 50, 50 # kara deliğe düşerse başa dön if player.colliderect(trophy): font = pygame.font.SysFont(None, 55) text = font.render("Kazandınız!", True, GOLD) screen.blit(text, (WIDTH//2 - 100, HEIGHT//2)) pygame.display.flip() pygame.time.wait(2000) running = False pygame.display.flip()
